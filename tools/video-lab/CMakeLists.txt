cmake_minimum_required(VERSION 3.20)
project(video_lab VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil libavfilter libswscale)
find_package(PNG REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
set(NINFER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_library(video_pipeline "${NINFER_ROOT}/src/media/local_video/video_pipeline.cpp")
target_include_directories(video_pipeline PUBLIC "${NINFER_ROOT}/src")
target_link_libraries(video_pipeline PUBLIC PkgConfig::AV)
target_compile_options(video_pipeline PRIVATE -Wall -Wextra -Wpedantic)
add_executable(video-lab src/main.cpp)
target_link_libraries(video-lab PRIVATE video_pipeline PNG::PNG nlohmann_json::nlohmann_json)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(video-lab-api-test tests/api_test.cpp)
  target_link_libraries(video-lab-api-test PRIVATE video_pipeline)
  add_test(NAME video_pipeline_behavior
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py
      $<TARGET_FILE:video-lab> $<TARGET_FILE:video-lab-api-test>)
endif()
